Reading the sketch

const diameter = 150;
let col;                    // declared globally, no value yet — color() isn't callable here
const bg = 230;

function setup() {
    createCanvas(640, 360);
    background(bg);

    col = color(255, 0, 100, 150);   // safe to call color() now that setup() is running
}

function draw() {
}

function mousePressed() {
    noStroke();
    fill(col);                        // was fill(r, g, b, opacity)
    circle(mouseX, mouseY, diameter);
}

function keyPressed() {
    background(bg);
}

The four separate colour numbers are gone; col — one color() object — replaces all four, and fill() now takes col directly instead of four arguments.
